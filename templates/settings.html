📖 Примеры использования на своём сайте 
// JavaScript — с любого сайта (базовый адрес как у OpenAI):
const BASE = "https://help.reallyvisuals.cloudpub.ru/api/v1";
const r = await fetch(BASE + "/ping?url=https://example.com",
  {headers:{"X-NetPulse-Key":"sp-test-ВАШ_КЛЮЧ"}});
const j = await r.json();  // {ok:true, reachable:true, latency_ms:42.1}

// curl / Python:
//   GET /api/v1/status?url=https://site.ru     — up/down, HTTP-код, задержка
//   GET /api/v1/speedtest/start?bytes=10&downloads=4&upload=true → id; затем
//   GET /api/v1/speedtest/result?id=…          — СПИДТЕСТ через наши серверы (↓/↑ Мбит/с, пинг)
//   GET /api/v1/speed?bytes=10&downloads=4     — быстрый синхронный спидтест
//   GET /api/v1/latency?host=site.ru&n=5       — avg/min/max/jitter задержки до узла
//   GET /api/v1/trace?url=https://site.ru      — трассировка пути через CDN edge
//   GET /api/v1/dns?host=site.ru               — A/AAAA записи
//   GET /api/v1/whois?domain=site.ru           — данные регистратора
//   GET /api/v1/search?q=новости&mode=web      — Google Custom Search
//   GET /api/v1/subnet?cidr=10.0.0.0/24        — разбор подсети
//   GET /api/v1/units?v=100&from=mbps&to=auto  — конвертер единиц
//   GET /api/v1/ip?geo=1                       — IP клиента + гео
//   GET /api/v1/uptime                         — ваши мониторы и uptime%
//   POST /api/v1/uptime  {url, interval_sec}   — добавить монитор
//   GET /api/v1/info                           — инфо о ключе
